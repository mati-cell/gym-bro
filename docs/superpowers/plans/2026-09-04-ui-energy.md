```ts
export function countThisWeek(
  sessions: Array<{ session_date: string }>,
  now: Date = new Date(),
): number {
  const start = new Date(now);
  start.setHours(0, 0, 0, 0);
  start.setDate(start.getDate() - 6);
  return sessions.filter((s) => {
    const d = new Date(`${s.session_date}T00:00:00`);
    return d >= start && d <= now;
  }).length;
}

export function sessionVolume(
  exercises: Array<{
    planned_weight_kg: number | null;
    planned_sets: number | null;
    planned_reps: string | null;
    performed: boolean;
  }>,
): number {
  let total = 0;
  for (const ex of exercises) {
    if (!ex.performed) continue;
    const m = ex.planned_reps?.match(/\d+/);
    const reps = m ? Number.parseInt(m[0], 10) : 0;
    if (ex.planned_weight_kg == null || ex.planned_sets == null || !reps) continue;
    total += ex.planned_weight_kg * ex.planned_sets * reps;
  }
  return Math.round(total);
}

export function actionTone(action: string | null): string {
  switch (action) {
    case "subir_peso":
      return "bg-success/15 text-success";
    case "bajar_peso":
      return "bg-warning/15 text-warning";
    case "revisar_tecnica":
      return "bg-chart-4/15 text-chart-4";
    case "cambiar_ejercicio":
      return "bg-destructive/15 text-destructive";
    default:
      return "bg-muted text-muted-foreground";
  }
}
```

- [ ] **Step 1: Crear `src/components/CountdownRing.tsx`**

```tsx
import { useId } from "react";

const TONES = {
  active: "var(--color-primary)",
  urgent: "var(--color-warning)",
  done: "var(--color-success)",
} as const;

export function CountdownRing({
  progress,
  size = 220,
  stroke = 14,
  tone,
  children,
}: {
  progress: number;
  size?: number;
  stroke?: number;
  tone: keyof typeof TONES;
  children?: React.ReactNode;
}) {
  const id = useId();
  const p = Math.max(0, Math.min(1, progress));
  const r = (size - stroke) / 2;
  const c = 2 * Math.PI * r;
  return (
    <div
      role="progressbar"
      aria-valuenow={Math.round(p * 100)}
      aria-valuemin={0}
      aria-valuemax={100}
      style={{ width: size, height: size }}
      className={`relative ${tone === "urgent" ? "animate-pulse" : ""}`}
    >
      <svg width={size} height={size} className="-rotate-90">
        <circle
          cx={size / 2}
          cy={size / 2}
          r={r}
          fill="none"
          stroke="var(--color-muted)"
          strokeWidth={stroke}
        />
        <circle
          cx={size / 2}
          cy={size / 2}
          r={r}
          fill="none"
          stroke={TONES[tone]}
          strokeWidth={stroke}
          strokeLinecap="round"
          strokeDasharray={c}
          strokeDashoffset={c * (1 - p)}
        />
        <defs>
          <linearGradient id={id} x1="0" y1="0" x2="1" y2="1">
            <stop offset="0%" style={{ stopColor: "var(--color-primary)" }} />
            <stop offset="100%" style={{ stopColor: "var(--color-chart-2)" }} />
          </linearGradient>
        </defs>
      </svg>
      <div className="absolute inset-0 flex items-center justify-center">{children}</div>
    </div>
  );
}
```

Nota: el `defs`/`useId` queda para un degradado futuro; el frente usa color
sólido por tono (YAGNI: no cablear el gradiente hasta necesitarlo). Si
`useId`/`defs` generan warning de lint por variable no usada (`id` SÍ se usa
en el `id=` del gradiente, así que no hay unused).
